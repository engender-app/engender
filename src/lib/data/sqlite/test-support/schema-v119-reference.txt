# through dumpSchema() in this directory's schema-dump.ts. This is the ground
# truth the post-rc.1 squash is held to (pre-publish ticket 01, ADR-0006), and
# squash-v119.test.ts compares against it. Its rows twin is
# schema-v119-reference-rows.txt; the rc.1 released journal's rows after the
# same chain are in schema-v119-rc1-rows.txt and schema-v119-rc1-seeded-rows.txt.
#
# Source: the chain at fec24e701 (functional-audit after FA 50 and FA 55),
# the tip this squash branched from. The commit that adds this file still
# carries that chain unchanged, and its squash-v119.test.ts builds an empty
# database and the rc.1 released journal through every entry of it and
# compares the dumps with these files. To reproduce:
#
#   git checkout $(git log --diff-filter=A --format=%H -- <this file>)
#   npx vitest run src/lib/data/sqlite/squash-v119.test.ts
#
# The rc.1 journal, migrated by the same chain, dumps to this same text.
#
# This file is frozen. It was not derived from the squash and is never
# regenerated from it; a migration added after v119 changes the live schema
# and leaves this file alone.

view media_file on media_file
  CREATE VIEW media_file AS SELECT p.uuid AS id,p.file_path AS file_name,CASE WHEN p.milestone_id IS NULL THEN 'entry' ELSE 'milestone' END AS source,COALESCE(p.epoch_day_override,e.epoch_day,m.epoch_day)AS epoch_day,m.name AS owner_name,COALESCE(m.uuid,CAST(e.id AS TEXT))AS owner_id,p.starred AS starred,0 AS source_rank,p.order_index AS owner_order,p.id AS row_id,CASE WHEN e.trashed_at IS NULL THEN 1 ELSE 0 END AS visible FROM photo p LEFT JOIN entry e ON e.id = p.entry_id LEFT JOIN milestone m ON m.id = p.milestone_id UNION ALL SELECT h.uuid,h.file_path,'hair',h.epoch_day,NULL,h.uuid,0,1,0,h.id,1 FROM hair_photo h UNION ALL SELECT r.uuid,r.file_path,'hairRemoval',s.epoch_day,NULL,s.uuid,0,2,0,r.id,1 FROM hair_removal_photo r JOIN hair_removal_session s ON s.id = r.session_id UNION ALL SELECT t.uuid,t.file_path,'tryout',t.epoch_day,y.label,y.uuid,0,3,0,t.id,1 FROM tryout_photo t JOIN tryout y ON y.id = t.tryout_id UNION ALL SELECT c.uuid,c.file_path,'procedure',c.epoch_day,d.name,d.uuid,0,4,0,c.id,1 FROM procedure_photo c JOIN procedure d ON d.id = c.procedure_id UNION ALL SELECT v.uuid,v.file_path,'video',n.epoch_day,NULL,CAST(n.id AS TEXT),0,5,v.order_index,v.id,CASE WHEN n.trashed_at IS NULL THEN 1 ELSE 0 END FROM video_note v JOIN entry n ON n.id = v.entry_id UNION ALL SELECT a.uuid,a.file_path,'voiceRecording',NULL,NULL,CAST(a.entry_id AS TEXT),0,NULL,a.order_index,a.id,1 FROM voice_recording a UNION ALL SELECT d.uuid,d.file_path,'document',d.epoch_day,NULL,d.uuid,0,NULL,0,d.id,1 FROM document d UNION ALL SELECT b.uuid,b.passage_file_path,'voiceBenchmarkPassage',b.epoch_day,NULL,b.uuid,0,NULL,0,b.id,1 FROM voice_benchmark b UNION ALL SELECT b.uuid,b.vowel_file_path,'voiceBenchmarkVowel',b.epoch_day,NULL,b.uuid,0,NULL,0,b.id,1 FROM voice_benchmark b WHERE b.vowel_file_path IS NOT NULL
